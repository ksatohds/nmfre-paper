get_p_topic_given_word <- function(fit) {  K <- fit$settings$dim$K  beta <- exp(fit$beta$logbeta[[1]])  beta <- as.matrix(beta)  # 念のため向きを揃える（K x V）  if (nrow(beta) != K && ncol(beta) == K) beta <- t(beta)  stopifnot(nrow(beta) == K)  colnames(beta) <- fit$vocab  p_t_given_w <- sweep(beta, 2, colSums(beta), "/")  p_t_given_w}p_t_given_w <- get_p_topic_given_word(fit)get_rep_words <- function(p_t_given_w, topic_id, thr = 0.5, topn = 30) {  v <- p_t_given_w[topic_id, ]  idx <- which(v > thr)  if (length(idx) == 0) return(character(0))  idx <- idx[order(v[idx], decreasing = TRUE)]  words <- colnames(p_t_given_w)[idx]  if (length(words) > topn) words <- words[1:topn]  words}for (j in 1:K) {  k <- topic_perm[j]  rw <- get_rep_words(p_t_given_w, topic_id = k, thr = thr_repword, topn = topn_repword)  cat(sprintf("\n[Display Topic %d  (STM Topic %d)]  thr=%.2f\n", j, k, thr_repword))  if (length(rw) == 0) {    cat("  (no words exceed threshold)\n")  } else {    cat("  ", paste(rw, collapse = ", "), "\n", sep="")  }}
